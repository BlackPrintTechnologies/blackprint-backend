WITH municipality_boundary AS (
    SELECT 
        ST_Union(
            ST_SetSRID(ST_MakePoint(
                CAST(JSON_EXTRACT_PATH_TEXT(centroid, 'coordinates', '0') AS FLOAT),
                CAST(JSON_EXTRACT_PATH_TEXT(centroid, 'coordinates', '1') AS FLOAT)
            ), 4326)
        ) as geom,
        COUNT(*) as point_count,
        MAX(centroid) as sample_centroid,
        MAX(h3_indexes) as sample_h3_indexes
    FROM blackprint_db_prd.data_product.v_qro 
    WHERE cve_mun = '14'
    AND centroid IS NOT NULL
    LIMIT 1000
),
municipality_buffer AS (
    SELECT ST_Buffer(geom, 0.1) as geom, point_count, sample_centroid, sample_h3_indexes
    FROM municipality_boundary
),
h3_values AS (
    SELECT H3_Polyfill(geom, 10) AS h3_indexes, point_count, sample_centroid, sample_h3_indexes
    FROM municipality_buffer
),
h3_index AS (
    SELECT o AS h3_value, point_count, sample_centroid, sample_h3_indexes
    FROM h3_values i, i.h3_indexes o
)
SELECT 
    SUM(a.total_usuarios_unicos) as total_users,
    COUNT(DISTINCT b.h3_value) as h3_count,
    MAX(b.point_count) as municipality_points,
    MAX(b.sample_centroid) as sample_centroid,
    MAX(b.sample_h3_indexes) as sample_h3_indexes,
    COUNT(*) as joined_records
FROM blackprint_db_prd.staging.stg_data_movilidad_por_hora_qro a
INNER JOIN h3_index b ON a.h3_index::VARCHAR = b.h3_value::VARCHAR
WHERE a.tipo_usuario = 'peaton'
